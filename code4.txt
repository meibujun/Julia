module ReactionNormModel

using LinearAlgebra

"""
    ReactionNormOptions

Options for reaction norm model fitting.
"""
@kwdef struct ReactionNormOptions
    center_environments::Bool = true
    scale_environments::Bool = true
    correlation_structure::Symbol = :unstructured  # :unstructured, :zero, :fixed
    fixed_correlation::Float64 = 0.0
    include_heterogeneous_residuals::Bool = false
    n_environmental_classes::Int = 1
    solver::Symbol = :direct
    max_iterations::Int = 100
    tolerance::Float64 = CONVERGENCE_TOL
end

"""
    fit_reaction_norm(population::Population; 
                     options::ReactionNormOptions=ReactionNormOptions(),
                     fixed_effects::Union{AbstractMatrix, Nothing}=nothing)

Fit reaction norm model for G×E interactions.
"""
function fit_reaction_norm(population::Population; 
                          options::ReactionNormOptions=ReactionNormOptions(),
                          fixed_effects::Union{AbstractMatrix, Nothing}=nothing)
    
    @info "Fitting reaction norm model"
    
    n = population.genotype_data.n_individuals
    p = size(population.phenotypes, 2)
    
    # Check for environmental data
    if size(population.environments, 2) == 0
        error("No environmental data provided for reaction norm model")
    end
    
    # Process environmental covariates
    E = process_environments(population.environments, options)
    
    # Build GRM
    @info "Computing genomic relationship matrix"
    geno_data = population.genotype_data
    update_allele_frequencies!(geno_data)
    X_geno, _ = noia_encode(geno_data.genotypes, geno_data.allele_freq)
    G = compute_grm(X_geno, geno_data.allele_freq)
    
    # Set up fixed effects
    X = setup_fixed_effects_rn(n, E, fixed_effects, options)
    
    # Fit model for each trait
    if p == 1
        model = fit_single_trait_reaction_norm(
            vec(population.phenotypes), X, G, E, options
        )
    else
        model = fit_multi_trait_reaction_norm(
            population.phenotypes, X, G, E, options
        )
    end
    
    # Create model info
    model_info = ModelInfo(
        traits = ["Trait$i" for i in 1:p],
        n_observations = n,
        n_fixed_effects = size(X, 2),
        n_random_effects = Dict(
            :intercept => n,
            :slope => n * size(E, 2)
        ),
        fit_date = now(),
        julia_version = VERSION,
        package_version = PACKAGE_VERSION,
        computation_time = 0.0,
        options = Dict(pairs(options))
    )
    
    # Extract environment range for predictions
    env_range = (minimum(E), maximum(E))
    
    return ReactionNormGBLUP(
        G,
        X,
        model.fixed_coef,
        model.intercepts,
        model.slopes,
        model.var_intercept,
        model.var_slope,
        model.cov_int_slope,
        model.residual_var,
        env_range,
        model_info
    )
end

"""
    fit_single_trait_reaction_norm(y, X, G, E, options)

Fit single-trait reaction norm model.
"""
function fit_single_trait_reaction_norm(y::AbstractVector{<:Real},
                                       X::AbstractMatrix{<:Real},
                                       G::AbstractMatrix{<:Real},
                                       E::AbstractMatrix{<:Real},
                                       options::ReactionNormOptions)
    
    n = length(y)
    n_env = size(E, 2)
    
    # Design matrices for random regression
    Z0 = Matrix(1.0I, n, n)  # For intercepts
    Z1 = E                   # For slopes (can be multivariate)
    
    # Combined design matrix
    Z = [Z0 Z1]
    
    # Covariance structure for random effects
    if options.correlation_structure == :unstructured
        # Estimate full covariance matrix
        result = fit_random_regression_reml(y, X, Z, G, n_env + 1)
    elseif options.correlation_structure == :zero
        # Independent intercepts and slopes
        result = fit_random_regression_independent(y, X, Z, G, n_env + 1)
    else
        # Fixed correlation
        result = fit_random_regression_fixed_cor(y, X, Z, G, n_env + 1, 
                                                options.fixed_correlation)
    end
    
    # Extract components
    fixed_coef = result.β
    random_effects = result.u
    
    # Split random effects into intercepts and slopes
    intercepts = random_effects[1:n]
    slopes = reshape(random_effects[(n+1):end], n, n_env)
    
    # Variance components
    Σ = result.variance_components
    var_intercept = [Σ[1, 1]]
    var_slope = [Σ[2, 2]]  # Simplified for univariate environment
    cov_int_slope = [Σ[1, 2]]
    
    # Residual variance
    residual_var = [result.σ²_e]
    
    return (
        fixed_coef = fixed_coef,
        intercepts = reshape(intercepts, :, 1),
        slopes = slopes,
        var_intercept = var_intercept,
        var_slope = var_slope,
        cov_int_slope = cov_int_slope,
        residual_var = residual_var
    )
end

"""
    fit_random_regression_reml(y, X, Z, G, q)

Fit random regression model using REML.
"""
function fit_random_regression_reml(y::AbstractVector{<:Real},
                                   X::AbstractMatrix{<:Real},
                                   Z::AbstractMatrix{<:Real},
                                   G::AbstractMatrix{<:Real},
                                   q::Int)  # Number of random coefficients
    
    n = length(y)
    p = size(X, 2)
    
    # Initial variance components
    var_y = var(y)
    Σ_init = Matrix(var_y/q * I, q, q)
    σ²_e_init = var_y / 2
    
    # Parameter vector: upper triangle of Σ and σ²_e
    n_params = div(q * (q + 1), 2) + 1
    θ_init = zeros(n_params)
    
    # Fill initial values
    idx = 1
    for i in 1:q
        for j in i:q
            θ_init[idx] = Σ_init[i, j]
            idx += 1
        end
    end
    θ_init[end] = σ²_e_init
    
    # Objective function
    function neg_log_likelihood(θ::Vector{T}) where T
        # Reconstruct Σ
        Σ = zeros(T, q, q)
        idx = 1
        for i in 1:q
            for j in i:q
                Σ[i, j] = θ[idx]
                Σ[j, i] = θ[idx]
                idx += 1
            end
        end
        σ²_e = θ[end]
        
        # Check positive definiteness
        if !isposdef(Σ) || σ²_e <= 0
            return T(Inf)
        end
        
        # Construct covariance matrix
        # V = ZGΣ(ZG)' + σ²_e I
        K = kron(Σ, G)  # Kronecker product
        V = Z * K * Z' + σ²_e * I
        
        # Add ridge for stability
        V += RIDGE_LAMBDA * I
        
        # Compute log-likelihood
        try
            L = cholesky(Symmetric(V))
            
            # Project out fixed effects
            V_inv_X = L \ X
            XtVinvX = X' * V_inv_X
            XtVinvX_inv = inv(XtVinvX)
            
            # Log-likelihood components
            logdet_V = 2 * sum(log.(diag(L.L)))
            logdet_XtVinvX = logdet(XtVinvX)
            
            V_inv_y = L \ y
            XtVinv_y = X' * V_inv_y
            y_Py = dot(y, V_inv_y) - dot(XtVinv_y, XtVinvX_inv * XtVinv_y)
            
            # REML log-likelihood (negative)
            loglik = 0.5 * (logdet_V + logdet_XtVinvX + y_Py)
            
            return loglik
        catch
            return T(Inf)
        end
    end
    
    # Optimize
    result = optimize(neg_log_likelihood, θ_init, LBFGS(),
                     Optim.Options(iterations=100, g_tol=1e-6))
    
    # Extract optimal parameters
    θ_opt = Optim.minimizer(result)
    
    # Reconstruct Σ
    Σ_opt = zeros(q, q)
    idx = 1
    for i in 1:q
        for j in i:q
            Σ_opt[i, j] = θ_opt[idx]
            Σ_opt[j, i] = θ_opt[idx]
            idx += 1
        end
    end
    σ²_e_opt = θ_opt[end]
    
    # Compute solutions
    K = kron(Σ_opt, G)
    V = Z * K * Z' + σ²_e_opt * I + RIDGE_LAMBDA * I
    
    V_inv = inv(V)
    
    # Fixed effects
    XtVinvX = X' * V_inv * X
    XtVinv_y = X' * V_inv * y
    β = XtVinvX \ XtVinv_y
    
    # Random effects (BLUPs)
    resid = y - X * β
    u = K * Z' * V_inv * resid
    
    return (
        β = β,
        u = u,
        variance_components = Σ_opt,
        σ²_e = σ²_e_opt
    )
end

"""
    predict_reaction_norm(model::ReactionNormGBLUP, environment::Real;
                         individuals::Union{Vector{Int}, Nothing}=nothing)

Predict breeding values for specific environment.
"""
function predict_reaction_norm(model::ReactionNormGBLUP, environment::Real;
                              individuals::Union{Vector{Int}, Nothing}=nothing)
    
    n = size(model.intercepts, 1)
    p = size(model.intercepts, 2)
    
    # Select individuals
    ind_idx = isnothing(individuals) ? (1:n) : individuals
    
    # Standardize environment if needed
    env_std = (environment - model.environments_range[1]) / 
              (model.environments_range[2] - model.environments_range[1])
    
    # Compute environment-specific GEBVs
    gebv = zeros(length(ind_idx), p)
    
    for trait in 1:p
        gebv[:, trait] = model.intercepts[ind_idx, trait] .+ 
                        model.slopes[ind_idx, trait] .* env_std
    end
    
    return gebv
end

"""
    compute_environmental_sensitivity(model::ReactionNormGBLUP;
                                    trait::Int=1)

Compute environmental sensitivity (plasticity) metrics.
"""
function compute_environmental_sensitivity(model::ReactionNormGBLUP;
                                         trait::Int=1)
    
    n = size(model.intercepts, 1)
    
    # Slope magnitude (absolute plasticity)
    plasticity = abs.(model.slopes[:, trait])
    
    # Relative plasticity (slope / intercept)
    relative_plasticity = abs.(model.slopes[:, trait] ./ 
                              (model.intercepts[:, trait] .+ 1e-10))
    
    # Variance of predicted values across environments
    env_range = range(model.environments_range[1], 
                     model.environments_range[2], 
                     length=100)
    
    gebv_variance = zeros(n)
    for i in 1:n
        gebv_across_env = [predict_reaction_norm(model, e, 
                                               individuals=[i])[1, trait] 
                          for e in env_range]
        gebv_variance[i] = var(gebv_across_env)
    end
    
    # Rank correlation across environments
    rank_stability = compute_rank_stability(model, trait, env_range)
    
    return Dict(
        :plasticity => plasticity,
        :relative_plasticity => relative_plasticity,
        :gebv_variance => gebv_variance,
        :rank_stability => rank_stability,
        :mean_plasticity => mean(plasticity),
        :cv_plasticity => std(plasticity) / mean(plasticity)
    )
end

"""
    stability_analysis(model::ReactionNormGBLUP;
                      n_environments::Int=10,
                      trait::Int=1)

Perform stability analysis across environments.
"""
function stability_analysis(model::ReactionNormGBLUP;
                          n_environments::Int=10,
                          trait::Int=1)
    
    n = size(model.intercepts, 1)
    
    # Define environment grid
    env_grid = range(model.environments_range[1],
                    model.environments_range[2],
                    length=n_environments)
    
    # Compute GEBVs across environments
    gebv_matrix = zeros(n, n_environments)
    for (j, env) in enumerate(env_grid)
        gebv_matrix[:, j] = predict_reaction_norm(model, env)[:, trait]
    end
    
    # Stability metrics
    
    # 1. Finlay-Wilkinson regression
    fw_slopes = zeros(n)
    fw_r2 = zeros(n)
    env_means = mean(gebv_matrix, dims=1)[:]
    
    for i in 1:n
        X_fw = [ones(n_environments) env_means]
        y_fw = gebv_matrix[i, :]
        β_fw = X_fw \ y_fw
        fw_slopes[i] = β_fw[2]
        
        y_pred = X_fw * β_fw
        ss_tot = sum((y_fw .- mean(y_fw)).^2)
        ss_res = sum((y_fw .- y_pred).^2)
        fw_r2[i] = 1 - ss_res / ss_tot
    end
    
    # 2. Wricke's ecovalence
    ecovalence = zeros(n)
    grand_mean = mean(gebv_matrix)
    genotype_means = mean(gebv_matrix, dims=2)[:]
    env_effects = mean(gebv_matrix, dims=1)[:] .- grand_mean
    
    for i in 1:n
        for j in 1:n_environments
            gxe = gebv_matrix[i, j] - genotype_means[i] - env_effects[j] - grand_mean
            ecovalence[i] += gxe^2
        end
    end
    
    # 3. Shukla's stability variance
    shukla_variance = ecovalence * (n - 1) / ((n - 1) * (n_environments - 2))
    
    # 4. Environmental correlation matrix
    env_cor = cor(gebv_matrix')
    
    # 5. Identify stable genotypes
    stable_genotypes = findall((fw_slopes .≈ 1.0) .& (fw_r2 .> 0.9))
    plastic_genotypes = findall(abs.(fw_slopes .- 1.0) .> 0.2)
    
    return Dict(
        :gebv_matrix => gebv_matrix,
        :environments => collect(env_grid),
        :fw_slopes => fw_slopes,
        :fw_r2 => fw_r2,
        :ecovalence => ecovalence,
        :shukla_variance => shukla_variance,
        :env_correlation => env_cor,
        :stable_genotypes => stable_genotypes,
        :plastic_genotypes => plastic_genotypes,
        :mean_stability => mean(fw_r2),
        :gxe_variance => var(vec(gebv_matrix))
    )
end

function process_environments(E::AbstractMatrix, options::ReactionNormOptions)
    E_processed = copy(E)
    
    if options.center_environments
        E_processed .-= mean(E_processed, dims=1)
    end
    
    if options.scale_environments
        E_processed ./= std(E_processed, dims=1)
    end
    
    return E_processed
end

"""
    compute_rank_stability(model::ReactionNormGBLUP, trait::Int, env_range)

Compute ranking stability across environments.
"""
function compute_rank_stability(model::ReactionNormGBLUP, trait::Int, env_range)
    n = size(model.intercepts, 1)
    n_env = length(env_range)
    
    # Get rankings at each environment
    rankings = zeros(Int, n, n_env)
    
    for (j, env) in enumerate(env_range)
        gebv = predict_reaction_norm(model, env)[:, trait]
        rankings[:, j] = invperm(sortperm(gebv, rev=true))
    end
    
    # Compute Spearman correlations between environments
    rank_cors = cor(rankings')
    
    # Average correlation (stability metric)
    upper_tri = [rank_cors[i, j] for i in 1:(n_env-1) for j in (i+1):n_env]
    mean_stability = mean(upper_tri)
    
    return mean_stability
end

"""
    fit_random_regression_independent(y, X, Z, G, q)

Fit random regression with independent variance components.
"""
function fit_random_regression_independent(y::AbstractVector{<:Real},
                                         X::AbstractMatrix{<:Real},
                                         Z::AbstractMatrix{<:Real},
                                         G::AbstractMatrix{<:Real},
                                         q::Int)
    
    # Simplified version assuming diagonal Σ
    n = length(y)
    
    # Initial values
    σ²_vec = fill(var(y) / q, q)
    σ²_e = var(y) / 2
    
    # EM algorithm
    max_iter = 50
    for iter in 1:max_iter
        # E-step: compute expected values
        Σ = Diagonal(σ²_vec)
        K = kron(Σ, G)
        V = Z * K * Z' + σ²_e * I + RIDGE_LAMBDA * I
        
        V_inv = inv(V)
        
        # Fixed effects
        β = inv(X' * V_inv * X) * X' * V_inv * y
        resid = y - X * β
        
        # Random effects
        u = K * Z' * V_inv * resid
        
        # M-step: update variance components
        u_split = reshape(u, size(G, 1), q)
        G_inv = inv(G + RIDGE_LAMBDA * I)
        
        for i in 1:q
            u_i = u_split[:, i]
            σ²_vec[i] = dot(u_i, G_inv * u_i) / size(G, 1)
        end
        
        # Update residual variance
        e = resid - Z * u
        σ²_e = dot(e, e) / n
        
        # Check convergence
        # (simplified)
    end
    
    return (
        β = β,
        u = u,
        variance_components = Diagonal(σ²_vec),
        σ²_e = σ²_e
    )
end

module MatingOptimization

    MatingConstraints

Constraints for mating optimization.
"""
@kwdef struct MatingConstraints
    max_matings_per_male::Int = 50
    max_matings_per_female::Int = 2
    min_matings_per_parent::Int = 0
    avoid_inbreeding_threshold::Float64 = 0.25
    maintain_diversity::Bool = true

    enforce_negative_assortative::Bool = false

"""
    optimal_mating(model::OGBLUP, population::Population;
                  strategy::MatingStrategy=OptimalContributionSelection(0.01),
                  constraints::MatingConstraints=MatingConstraints(),
                  sex_info::Union{Vector{String}, Nothing}=nothing)

Design optimal mating plan considering all genetic effects.
"""
function optimal_mating(model::OGBLUP, population::Population;
                       strategy::MatingStrategy=OptimalContributionSelection(0.01),
                       constraints::MatingConstraints=MatingConstraints(),
                       sex_info::Union{Vector{String}, Nothing}=nothing)
    
    @info "Optimizing mating plan using $(typeof(strategy))"
    
    n = population.genotype_data.n_individuals
    
    # Determine sex of individuals
    if isnothing(sex_info)
        # Assume random sex assignment if not provided
        sex_info = [i <= div(n, 2) ? "M" : "F" for i in 1:n]
    end
    
    males = findall(sex_info .== "M")
    females = findall(sex_info .== "F")
    
    @info "Found $(length(males)) males and $(length(females)) females"
    
    # Get breeding values
    bv = compute_breeding_values(model)
    
    # Apply mating strategy
    if isa(strategy, RandomMating)
        mating_plan = random_mating_plan(males, females, constraints)
    elseif isa(strategy, MinimizeInbreeding)
        mating_plan = minimize_inbreeding_plan(males, females, model.grms.G, constraints)
    elseif isa(strategy, MaximizeGeneticGain)
        mating_plan = maximize_gain_plan(males, females, bv, model, constraints)
    elseif isa(strategy, OptimalContributionSelection)
        mating_plan = optimal_contribution_plan(males, females, bv, model, 
                                              strategy.target_inbreeding, constraints)
    else
        error("Unknown mating strategy: $(typeof(strategy))")
    end
    
    # Evaluate mating plan
    evaluation = evaluate_mating_plan(mating_plan, model, population)
    
    return MatingPlan(
        mating_plan.matings,
        evaluation.expected_merit,
        evaluation.expected_inbreeding,
        evaluation.diversity_score,
        mating_plan.constraints_satisfied,
        Dict(
            :strategy => string(typeof(strategy)),
            :n_males => length(males),
            :n_females => length(females),
            :evaluation => evaluation
        )
    )
end

"""
    random_mating_plan(males, females, constraints)

Generate random mating plan respecting constraints.
"""
function random_mating_plan(males::Vector{Int}, 
                          females::Vector{Int},
                          constraints::MatingConstraints)
    
    matings = Tuple{Int, Int}[]
    male_counts = zeros(Int, length(males))
    female_counts = zeros(Int, length(females))
    
    for _ in 1:constraints.total_matings
        # Find eligible males
        eligible_males = findall(male_counts .< constraints.max_matings_per_male)
        if isempty(eligible_males)
            break
        end
        
        # Find eligible females
        eligible_females = findall(female_counts .< constraints.max_matings_per_female)
        if isempty(eligible_females)
            break
        end
        
        # Random selection
        m_idx = rand(eligible_males)
        f_idx = rand(eligible_females)
        
        male = males[m_idx]
        female = females[f_idx]
        
        push!(matings, (male, female))
        male_counts[m_idx] += 1
        female_counts[f_idx] += 1
    end
    
    return (
        matings = matings,
        constraints_satisfied = length(matings) == constraints.total_matings
    )
end

"""
    minimize_inbreeding_plan(males, females, G, constraints)

Design matings to minimize average inbreeding.
"""
function minimize_inbreeding_plan(males::Vector{Int},
                                females::Vector{Int},
                                G::AbstractMatrix{<:Real},
                                constraints::MatingConstraints)
    
    n_males = length(males)
    n_females = length(females)
    
    # Create optimization model
    model = Model(GLPK.Optimizer)
    
    # Decision variables: x[i,j] = number of matings between male i and female j
    @variable(model, 0 <= x[1:n_males, 1:n_females] <= 1, Int)
    
    # Objective: minimize average relationship
    relationship_matrix = zeros(n_males, n_females)
    for i in 1:n_males
        for j in 1:n_females
            relationship_matrix[i, j] = G[males[i], females[j]]
        end
    end
    
    @objective(model, Min, sum(relationship_matrix .* x))
    
    # Constraints
    # Total matings
    @constraint(model, sum(x) == constraints.total_matings)
    
    # Male mating limits
    for i in 1:n_males
        @constraint(model, sum(x[i, :]) <= constraints.max_matings_per_male)
    end
    
    # Female mating limits
    for j in 1:n_females
        @constraint(model, sum(x[:, j]) <= constraints.max_matings_per_female)
    end
    
    # Avoid high relationships
    if constraints.avoid_inbreeding_threshold < 1.0
        for i in 1:n_males
            for j in 1:n_females
                if relationship_matrix[i, j] > constraints.avoid_inbreeding_threshold
                    @constraint(model, x[i, j] == 0)
                end
            end
        end
    end
    
    # Solve
    optimize!(model)
    
    # Extract solution
    matings = Tuple{Int, Int}[]
    x_sol = value.(x)
    
    for i in 1:n_males
        for j in 1:n_females
            if x_sol[i, j] > 0.5
                push!(matings, (males[i], females[j]))
            end
        end
    end
    
    return (
        matings = matings,
        constraints_satisfied = termination_status(model) == MOI.OPTIMAL
    )
end

"""
    maximize_gain_plan(males, females, bv, model, constraints)

Design matings to maximize genetic gain.
"""
function maximize_gain_plan(males::Vector{Int},
                          females::Vector{Int},
                          bv::Dict{Symbol, Any},
                          model::OGBLUP,
                          constraints::MatingConstraints)
    
    n_males = length(males)
    n_females = length(females)
    
    # Get breeding values
    male_bv = bv[:EBV][males, 1]  # Assuming first trait
    female_bv = bv[:EBV][females, 1]
    
    # Include dominance if available
    use_dominance = haskey(bv, :DGV)
    if use_dominance
        male_dgv = bv[:DGV][males, 1]
        female_dgv = bv[:DGV][females, 1]
    end
    
    # Expected progeny merit
    progeny_merit = zeros(n_males, n_females)
    for i in 1:n_males
        for j in 1:n_females
            # Additive contribution
            progeny_merit[i, j] = 0.5 * (male_bv[i] + female_bv[j])
            
            # Dominance contribution (simplified)
            if use_dominance
                # Expected heterozygosity boost
                progeny_merit[i, j] += 0.1 * abs(male_dgv[i] - female_dgv[j])
            end
        end
    end
    
    # Optimization model
    opt_model = Model(GLPK.Optimizer)
    
    @variable(opt_model, 0 <= x[1:n_males, 1:n_females] <= 1, Int)
    
    # Objective: maximize expected merit
    @objective(opt_model, Max, sum(progeny_merit .* x))
    
    # Standard constraints
    @constraint(opt_model, sum(x) == constraints.total_matings)
    
    for i in 1:n_males
        @constraint(opt_model, sum(x[i, :]) <= constraints.max_matings_per_male)
    end
    
    for j in 1:n_females
        @constraint(opt_model, sum(x[:, j]) <= constraints.max_matings_per_female)
    end
    
    # Diversity constraint if requested
    if constraints.maintain_diversity
        # Limit matings between top individuals
        top_males = sortperm(male_bv, rev=true)[1:min(5, n_males)]
        top_females = sortperm(female_bv, rev=true)[1:min(5, n_females)]
        
        for i in top_males
            for j in top_females
                @constraint(opt_model, x[i, j] <= 0.2 * constraints.total_matings / 25)
            end
        end
    end
    
    # Solve
    optimize!(opt_model)
    
    # Extract solution
    matings = Tuple{Int, Int}[]
    x_sol = value.(x)
    
    for i in 1:n_males
        for j in 1:n_females
            n_matings = round(Int, x_sol[i, j])
            for _ in 1:n_matings
                push!(matings, (males[i], females[j]))
            end
        end
    end
    
    return (
        matings = matings,
        constraints_satisfied = termination_status(opt_model) == MOI.OPTIMAL
    )
end

"""
    optimal_contribution_plan(males, females, bv, model, 
                            target_inbreeding, constraints)

Optimal contribution selection balancing gain and inbreeding.
"""
function optimal_contribution_plan(males::Vector{Int},
                                 females::Vector{Int},
                                 bv::Dict{Symbol, Any},
                                 model::OGBLUP,
                                 target_inbreeding::Float64,
                                 constraints::MatingConstraints)
    
    n = length(males) + length(females)
    all_individuals = vcat(males, females)
    
    # Get additive relationship matrix
    A = model.grms.G[all_individuals, all_individuals]
    
    # Get breeding values
    ebv = bv[:EBV][all_individuals, 1]
    
    # Optimal contribution problem
    # maximize: c'EBV - λ/2 * c'Ac
    # subject to: sum(c) = 1, c >= 0
    
    # Find optimal λ using bisection
    λ_min = 0.0
    λ_max = 1000.0
    
    for _ in 1:20
        λ = (λ_min + λ_max) / 2
        
        # Solve for contributions
        Q = A + λ * I
        c = Q \ ebv
        c = max.(c, 0.0)  # Non-negative
        c /= sum(c)  # Sum to 1
        
        # Check constraint
        ΔF = 0.5 * dot(c, A * c)
        
        if ΔF < target_inbreeding
            λ_max = λ
        else
            λ_min = λ
        end
    end
    
    # Final contributions
    Q = A + λ_max * I
    c = Q \ ebv
    c = max.(c, 0.0)
    c /= sum(c)
    
    # Convert contributions to matings
    # Split back into males and females
    n_males = length(males)
    c_males = c[1:n_males]
    c_females = c[(n_males+1):end]
    
    # Normalize within sex
    c_males /= sum(c_males)
    c_females /= sum(c_females)
    
    # Allocate matings proportionally
    matings = Tuple{Int, Int}[]
    
    for i in 1:length(males)
        n_matings_male = round(Int, c_males[i] * constraints.total_matings)
        
        # Distribute among females proportionally
        female_probs = c_females / sum(c_females)
        
        for _ in 1:n_matings_male
            # Sample female based on contributions
            j = sample(1:length(females), Weights(female_probs))
            push!(matings, (males[i], females[j]))
        end
    end
    
    # Ensure we have exactly the requested number
    while length(matings) < constraints.total_matings
        i = sample(1:length(males), Weights(c_males))
        j = sample(1:length(females), Weights(c_females))
        push!(matings, (males[i], females[j]))
    end
    
    while length(matings) > constraints.total_matings
        pop!(matings)
    end
    
    return (
        matings = matings,
        constraints_satisfied = true,
        contributions = c
    )
end

"""
    evaluate_mating_plan(mating_plan, model, population)

Evaluate expected outcomes of mating plan.
"""
function evaluate_mating_plan(mating_plan, model::OGBLUP, population::Population)
    
    n_matings = length(mating_plan.matings)
    
    # Initialize metrics
    expected_merit = zeros(n_matings)
    expected_inbreeding = zeros(n_matings)
    expected_heterozygosity = zeros(n_matings)
    
    # Get genetic values
    bv = compute_breeding_values(model)
    G = model.grms.G
    
    # Evaluate each mating
    for (k, (male, female)) in enumerate(mating_plan.matings)
        # Expected additive merit
        expected_merit[k] = 0.5 * (bv[:EBV][male, 1] + bv[:EBV][female, 1])
        
        # Expected inbreeding coefficient of progeny
        expected_inbreeding[k] = 0.5 * (1 + G[male, female])
        
        # Expected heterozygosity (simplified)
        # Based on difference in genotypes
        if haskey(population.metadata, :genotypes)
            geno_male = population.genotype_data.genotypes[male, :]
            geno_female = population.genotype_data.genotypes[female, :]
            
            # Proportion of loci where parents have different homozygotes
            diff_homo = sum((geno_male .== 0) .& (geno_female .== 2) .| 
                          (geno_male .== 2) .& (geno_female .== 0))
            expected_heterozygosity[k] = diff_homo / population.genotype_data.n_markers
        end
    end
    
    # Population-level metrics
    unique_parents = unique(vcat([m[1] for m in mating_plan.matings],
                                [m[2] for m in mating_plan.matings]))
    
    # Effective population size (simplified)
    n_males = length(unique([m[1] for m in mating_plan.matings]))
    n_females = length(unique([m[2] for m in mating_plan.matings]))
    Ne = 4 * n_males * n_females / (n_males + n_females)
    
    # Diversity metrics
    parent_relationships = G[unique_parents, unique_parents]
    diversity_score = 1.0 - mean(parent_relationships[triu(ones(Bool, size(parent_relationships)), 1)])
    
    return (
        expected_merit = expected_merit,
        expected_inbreeding = expected_inbreeding,
        expected_heterozygosity = expected_heterozygosity,
        mean_merit = mean(expected_merit),
        mean_inbreeding = mean(expected_inbreeding),
        diversity_score = diversity_score,
        effective_size = Ne,
        n_unique_males = n_males,
        n_unique_females = n_females
    )
end

"""
    simulate_offspring(mating_plan::MatingPlan, population::Population, 
                      model::OGBLUP; n_offspring_per_mating::Int=1)

Simulate offspring from mating plan.
"""
function simulate_offspring(mating_plan::MatingPlan, population::Population, 
                          model::OGBLUP; n_offspring_per_mating::Int=1)
    
    n_matings = length(mating_plan.matings)
    n_total_offspring = n_matings * n_offspring_per_mating
    n_markers = population.genotype_data.n_markers
    
    # Initialize offspring genotypes
    offspring_genotypes = zeros(Int8, n_total_offspring, n_markers)
    offspring_parents = Tuple{Int, Int}[]
    
    offspring_idx = 1
    for (male, female) in mating_plan.matings
        male_geno = population.genotype_data.genotypes[male, :]
        female_geno = population.genotype_data.genotypes[female, :]
        
        for _ in 1:n_offspring_per_mating
            # Simulate meiosis and fertilization
            for j in 1:n_markers
                # Sample one allele from each parent
                if male_geno[j] == 0
                    paternal_allele = 0
                elseif male_geno[j] == 2
                    paternal_allele = 1
                else  # Heterozygous
                    paternal_allele = rand() < 0.5 ? 0 : 1
                end
                
                if female_geno[j] == 0
                    maternal_allele = 0
                elseif female_geno[j] == 2
                    maternal_allele = 1
                else  # Heterozygous
                    maternal_allele = rand() < 0.5 ? 0 : 1
                end
                
                offspring_genotypes[offspring_idx, j] = paternal_allele + maternal_allele
            end
            
            push!(offspring_parents, (male, female))
            offspring_idx += 1
        end
    end
    
    # Create offspring population
    offspring_geno_data = GenotypeData(
        offspring_genotypes,
        copy(population.genotype_data.allele_freq),  # Will be updated
        population.genotype_data.maf_filter,
        ones(n_markers),  # Perfect call rate for simulated data
        n_total_offspring,
        n_markers,
        population.genotype_data.marker_info,
        DataFrame(ID = ["Offspring_$i" for i in 1:n_total_offspring],
                  Sire = [p[1] for p in offspring_parents],
                  Dam = [p[2] for p in offspring_parents]),
        population.genotype_data.ploidy
    )
    
    # Update allele frequencies
    update_allele_frequencies!(offspring_geno_data)
    
    # Predict phenotypes using the model
    predicted_phenotypes = predict_breeding_values_offspring(
        model, population, offspring_geno_data, offspring_parents
    )
    
    offspring_population = Population(
        offspring_geno_data,
        predicted_phenotypes,
        zeros(n_total_offspring, 0),  # No environmental data
        population.generation + 1,
        nothing,  # Could add pedigree
        nothing,  # No true values for real offspring
        Dict(:parents => offspring_parents)
    )
    
    return offspring_population
end

# Helper functions

"""
    predict_breeding_values_offspring(model, parent_pop, offspring_genos, parents)

Predict breeding values for offspring.
"""
function predict_breeding_values_offspring(model::OGBLUP,
                                         parent_pop::Population,
                                         offspring_genos::GenotypeData,
                                         parents::Vector{Tuple{Int, Int}})
    
    n_offspring = offspring_genos.n_individuals
    n_traits = size(model.fitted_values, 2)
    
    # Simple prediction: parent average plus Mendelian sampling
    predicted_bv = zeros(n_offspring, n_traits)
    
    parent_bv = predict_gebv(model, parent_pop, components=:additive)
    
    for (i, (sire, dam)) in enumerate(parents)
        # Parent average
        predicted_bv[i, :] = 0.5 * (parent_bv[sire, :] + parent_bv[dam, :])
        
        # Add Mendelian sampling variance
        # This is simplified - proper implementation would use segregation variance
        for trait in 1:n_traits
            σ_ms = sqrt(0.5 * model.var_comp.estimates.σ²_a)  # Approximate
            predicted_bv[i, trait] += randn() * σ_ms
        end
    end
    
    return predicted_bv
end

        for i in 1:n

            genotypes[i, j] = allele1 + allele2
        end
    end

    end

"""
    simulate_qtl_effects(scenario::SimulationScenario, base_pop::GenotypeData)

Simulate QTL positions and effects.
"""
function simulate_qtl_effects(scenario::SimulationScenario, base_pop::GenotypeData)
    m = base_pop.n_markers
    
    # Select QTL positions
    qtl_indices = sort(sample(1:m, scenario.n_qtl, replace=false))
    
    # Determine effect sizes based on genetic variance targets
    total_genetic_var = scenario.h2_additive + scenario.h2_dominance + scenario.h2_epistasis
    residual_var = 1.0 - total_genetic_var
    
    # Additive effects
    if scenario.qtl_distribution == :normal
        additive_effects = randn(scenario.n_qtl)
    elseif scenario.qtl_distribution == :gamma
        additive_effects = rand(Gamma(0.4, 1.0), scenario.n_qtl)
        additive_effects .*= rand([-1, 1], scenario.n_qtl)  # Random sign
    else  # exponential
        additive_effects = rand(Exponential(1.0), scenario.n_qtl)
        additive_effects .*= rand([-1, 1], scenario.n_qtl)
    end
    
    # Major QTL
    n_major = round(Int, scenario.major_qtl_proportion * scenario.n_qtl)
    major_qtl = sample(1:scenario.n_qtl, n_major, replace=false)
    additive_effects[major_qtl] .*= scenario.major_qtl_effect_size
    
    # Scale to achieve target variance
    scale_factor = sqrt(scenario.h2_additive / var(simulate_additive_values(base_pop, qtl_indices, additive_effects)))
    additive_effects .*= scale_factor
    
    # Dominance effects
    dominance_effects = zeros(scenario.n_qtl)
    if scenario.h2_dominance > 0
        # Dominance coefficients between -1 and 1
        dom_coefficients = 2 * rand(scenario.n_qtl) .- 1
        dominance_effects = dom_coefficients .* abs.(additive_effects)
        
        # Scale for target variance
        scale_factor_d = sqrt(scenario.h2_dominance / var(simulate_dominance_values(base_pop, qtl_indices, dominance_effects)))
        dominance_effects .*= scale_factor_d
    end
    
    # Epistatic pairs
    epistatic_pairs = Tuple{Int, Int}[]
    epistatic_effects = Float64[]
    
    if scenario.h2_epistasis > 0 && scenario.epistasis_type != :none
        if scenario.epistasis_type == :pairwise
            # Random pairs
            for _ in 1:scenario.n_epistatic_pairs
                q1 = rand(1:scenario.n_qtl)
                q2 = rand(setdiff(1:scenario.n_qtl, q1))
                push!(epistatic_pairs, (qtl_indices[q1], qtl_indices[q2]))
                
                # Epistatic effect proportional to main effects
                epi_effect = 0.5 * sqrt(abs(additive_effects[q1] * additive_effects[q2]))
                push!(epistatic_effects, rand([-1, 1]) * epi_effect)
            end
        elseif scenario.epistasis_type == :network
            # Hub-based network
            epistatic_pairs, epistatic_effects = simulate_epistatic_network(
                qtl_indices, additive_effects, scenario
            )
        end
        
        # Scale epistatic effects
        if !isempty(epistatic_effects)
            scale_factor_e = sqrt(scenario.h2_epistasis / var(simulate_epistatic_values(base_pop, epistatic_pairs, epistatic_effects)))
            epistatic_effects .*= scale_factor_e
        end
    end
    
    qtl_info = Dict(
        :indices => qtl_indices,
        :additive_effects => additive_effects,
        :dominance_effects => dominance_effects,
        :epistatic_pairs => epistatic_pairs,
        :epistatic_effects => epistatic_effects,
        :major_qtl => major_qtl,
        :allele_freq => base_pop.allele_freq[qtl_indices]
    )
    
    return qtl_info
end

"""
    calculate_phenotypes(geno_data::GenotypeData, qtl_info::Dict, scenario::SimulationScenario)

Calculate phenotypic values from genotypes and QTL effects.
"""
function calculate_phenotypes(geno_data::GenotypeData, qtl_info::Dict, scenario::SimulationScenario)
    n = geno_data.n_individuals
    phenotypes = zeros(n)
    
    # Additive genetic values
    for (i, qtl_idx) in enumerate(qtl_info[:indices])
        effect = qtl_info[:additive_effects][i]
        p = qtl_info[:allele_freq][i]
        
        # Centered genotype values
        for ind in 1:n
            g = geno_data.genotypes[ind, qtl_idx]
            phenotypes[ind] += effect * (g - 2*p)
        end
    end
    
    # Dominance deviations
    if scenario.h2_dominance > 0
        for (i, qtl_idx) in enumerate(qtl_info[:indices])
            d_effect = qtl_info[:dominance_effects][i]
            
            for ind in 1:n
                g = geno_data.genotypes[ind, qtl_idx]
                if g == 1  # Heterozygote
                    phenotypes[ind] += d_effect
                end
            end
        end
    end
    
    # Epistatic effects
    if scenario.h2_epistasis > 0
        for (k, (qtl1, qtl2)) in enumerate(qtl_info[:epistatic_pairs])
            epi_effect = qtl_info[:epistatic_effects][k]
            
            for ind in 1:n
                g1 = geno_data.genotypes[ind, qtl1]
                g2 = geno_data.genotypes[ind, qtl2]
                
                # Simple multiplicative epistasis
                phenotypes[ind] += epi_effect * (g1 - 1) * (g2 - 1)
            end
        end
    end
    
    # Add environmental noise
    genetic_var = var(phenotypes)
    if genetic_var > 0
        target_genetic_var = scenario.h2_additive + scenario.h2_dominance + scenario.h2_epistasis
        environmental_var = genetic_var * (1 - target_genetic_var) / target_genetic_var
        phenotypes .+= randn(n) * sqrt(environmental_var)
    end
    
    # Multiple traits if needed
    if scenario.n_environments > 1
        # Extend to multiple environments
        phenotypes = repeat(phenotypes, 1, scenario.n_environments)
        
        # Add G×E
        if scenario.gxe_variance > 0
            for env in 2:scenario.n_environments
                phenotypes[:, env] .+= randn(n) * sqrt(scenario.gxe_variance)
            end
        end
    end
    
    return reshape(phenotypes, n, :)
end

"""
    apply_selection(population::Population, scenario::SimulationScenario)

Apply selection to identify parents for next generation.
"""
function apply_selection(population::Population, scenario::SimulationScenario)
    n = population.genotype_data.n_individuals
    
    # Get selection criterion
    if scenario.selection_on_phenotype
        # Phenotypic selection
        criterion = vec(mean(population.phenotypes, dims=2))
    else
        # Genomic selection (would need to fit model)
        # For now, use phenotypes as proxy
        criterion = vec(mean(population.phenotypes, dims=2))
    end
    
    # Apply selection method
    if scenario.selection_method == :truncation
        # Simple truncation selection
        n_selected = round(Int, n * scenario.selection_intensity)
        selected = sortperm(criterion, rev=true)[1:n_selected]
        
    elseif scenario.selection_method == :optimal
        # Optimal contribution selection
        # Simplified version
        selected = optimal_contribution_selection(population, criterion, scenario)
        
    else  # :index
        # Index selection with multiple traits
        selected = index_selection(population, scenario)
    end
    
    return selected
end

"""
    simulate_generation(current_pop::Population, parents::Vector{Int},
                       qtl_info::Dict, scenario::SimulationScenario, gen::Int)

Simulate next generation from selected parents.
"""
function simulate_generation(current_pop::Population, parents::Vector{Int},
                           qtl_info::Dict, scenario::SimulationScenario, gen::Int)
    
    n_offspring = scenario.n_individuals
    n_markers = current_pop.genotype_data.n_markers
    
    # Determine matings
    if scenario.mating_system == :random
        matings = generate_random_matings(parents, n_offspring, scenario)
    elseif scenario.mating_system == :assortative
        matings = generate_assortative_matings(parents, current_pop, n_offspring)
    else  # :disassortative
        matings = generate_disassortative_matings(parents, current_pop, n_offspring)
    end
    
    # Generate offspring genotypes
    offspring_genotypes = zeros(Int8, n_offspring, n_markers)
    
    for (offspring_idx, (parent1, parent2)) in enumerate(matings)
        # Get parent genotypes
        p1_geno = current_pop.genotype_data.genotypes[parent1, :]
        p2_geno = current_pop.genotype_data.genotypes[parent2, :]
        
        # Simulate meiosis for each parent
        gamete1 = simulate_meiosis(p1_geno, current_pop.genotype_data.marker_info,
                                  scenario.chromosome_lengths, scenario.recombination_model)
        gamete2 = simulate_meiosis(p2_geno, current_pop.genotype_data.marker_info,
                                  scenario.chromosome_lengths, scenario.recombination_model)
        
        # Combine gametes
        offspring_genotypes[offspring_idx, :] = gamete1 .+ gamete2
        
        # Apply mutations if specified
        if scenario.mutation_rate > 0
            apply_mutations!(offspring_genotypes[offspring_idx, :], scenario.mutation_rate)
        end
    end
    
    # Update allele frequencies
    new_allele_freq = vec(mean(offspring_genotypes, dims=1)) / 2
    
    # Create new genotype data
    sample_info = DataFrame(
        ID = ["IND_$(gen)_$i" for i in 1:n_offspring],
        Sex = rand(["M", "F"], n_offspring),
        Population = ones(Int, n_offspring),
        Generation = fill(gen, n_offspring),
        Sire = [m[1] for m in matings],
        Dam = [m[2] for m in matings]
    )
    
    offspring_geno_data = GenotypeData(
        offspring_genotypes,
        new_allele_freq,
        current_pop.genotype_data.maf_filter,
        ones(n_markers),
        n_offspring,
        n_markers,
        current_pop.genotype_data.marker_info,
        sample_info,
        2
    )
    
    # Calculate new phenotypes
    new_phenotypes = calculate_phenotypes(offspring_geno_data, qtl_info, scenario)
    
    # Create new population
    new_population = Population(
        offspring_geno_data,
        new_phenotypes,
        zeros(n_offspring, size(current_pop.environments, 2)),
        gen,
        nothing,  # Pedigree could be added
        nothing,  # True values for simulation
        Dict(:parents => matings)
    )
    
    return new_population
end

function store_generation_results!(results::SimulationResults, 
                                 population::Population, 
                                 generation::Int)
    
    push!(results.generations, generation)
    
    # Phenotypic mean
    push!(results.mean_phenotype, vec(mean(population.phenotypes, dims=1)))
    
    # Genetic variance (simplified - would need proper decomposition)
    push!(results.genetic_variance, vec(var(population.phenotypes, dims=1)))
    
    # Allele frequencies
    if results.scenario.track_allele_frequencies
        push!(results.allele_frequencies, copy(population.genotype_data.allele_freq'))
    end
    
    # Inbreeding coefficient (from diagonal of GRM)
    # Simplified calculation
    push!(results.inbreeding, Float64[])
    
    # Effective population size
    if generation > 0 && !isempty(results.allele_frequencies)

    end

"""
    assign_markers_to_chromosomes(n_markers, n_chr, chr_lengths)

Assign markers to chromosomes.
"""
function assign_markers_to_chromosomes(n_markers::Int, n_chr::Int, chr_lengths::Vector{Float64})
    
    # Proportional assignment based on chromosome lengths
    total_length = sum(chr_lengths)
    markers_per_chr = [round(Int, n_markers * len / total_length) for len in chr_lengths]
    
    # Adjust for rounding
    while sum(markers_per_chr) < n_markers
        idx = argmin(markers_per_chr ./ chr_lengths)
        markers_per_chr[idx] += 1
    end
    while sum(markers_per_chr) > n_markers
        idx = argmax(markers_per_chr ./ chr_lengths)
        markers_per_chr[idx] -= 1
    end
    
    # Create marker info
    marker_data = []
    marker_idx = 1
    
    for (chr, (n_chr_markers, chr_len)) in enumerate(zip(markers_per_chr, chr_lengths))
        positions = sort(rand(n_chr_markers) * chr_len)
        
        for pos in positions
            push!(marker_data, (
                CHR = chr,
                POS = pos,
                ID = "SNP_$marker_idx",
                REF = "A",
                ALT = "T"
            ))
            marker_idx += 1
        end
    end
    
    return DataFrame(marker_data)
end

"""
    simulate_meiosis(genotype, marker_info, chr_lengths, recomb_model)

Simulate meiosis with recombination.
"""
function simulate_meiosis(genotype::Vector{Int8}, 
                        marker_info::DataFrame,
                        chr_lengths::Vector{Float64},
                        recomb_model::Symbol)
    
    n_markers = length(genotype)
    gamete = zeros(Int8, n_markers)
    
    # Process each chromosome
    for chr in unique(marker_info.CHR)
        chr_markers = findall(marker_info.CHR .== chr)
        if isempty(chr_markers)
            continue
        end
        
        # Determine number of crossovers (Poisson with rate = length/100)
        chr_length = chr_lengths[chr]
        n_crossovers = rand(Poisson(chr_length / 100))
        
        if n_crossovers == 0
            # No recombination, inherit from one chromosome
            inherited_strand = rand([0, 1])
            for idx in chr_markers
                gamete[idx] = (genotype[idx] >> inherited_strand) & 1
            end
        else
            # Generate crossover positions
            crossover_positions = sort(rand(n_crossovers) * chr_length)
            
            # Determine inheritance pattern
            current_strand = rand([0, 1])
            
            for (i, idx) in enumerate(chr_markers)
                marker_pos = marker_info.POS[idx]
                
                # Check if we've passed a crossover
                while !isempty(crossover_positions) && marker_pos > crossover_positions[1]
                    current_strand = 1 - current_strand
                    popfirst!(crossover_positions)
                end
                
                # Inherit from current strand
                if genotype[idx] == 0
                    gamete[idx] = 0
                elseif genotype[idx] == 2
                    gamete[idx] = 1
                else  # Heterozygote
                    gamete[idx] = current_strand
                end
            end
        end
    end
    
    return gamete
end

"""
    apply_mutations!(genotype, mutation_rate)

Apply mutations to a genotype.
"""
function apply_mutations!(genotype::Vector{Int8}, mutation_rate::Float64)
    n_markers = length(genotype)
    n_mutations = rand(Poisson(2 * n_markers * mutation_rate))
    
    for _ in 1:n_mutations
        pos = rand(1:n_markers)
        # Flip one allele
        if genotype[pos] == 0
            genotype[pos] = 1
        elseif genotype[pos] == 2
            genotype[pos] = 1
        else  # Heterozygote
            genotype[pos] = rand([0, 2])
        end
    end
end

"""
    generate_random_matings(parents, n_offspring, scenario)

Generate random matings from parent pool.
"""
function generate_random_matings(parents::Vector{Int}, n_offspring::Int, 
                               scenario::SimulationScenario)
    
    matings = Tuple{Int, Int}[]
    
    for _ in 1:n_offspring
        if scenario.selfing_rate > 0 && rand() < scenario.selfing_rate
            # Selfing
            parent = rand(parents)
            push!(matings, (parent, parent))
        else
            # Random mating
            p1 = rand(parents)
            p2 = rand(parents)
            while p1 == p2  # Avoid selfing if not allowed
                p2 = rand(parents)
            end
            push!(matings, (p1, p2))
        end
    end
    
    return matings
end

"""
    estimate_ne(freq1, freq2)

Estimate effective population size from allele frequency changes.
"""
function estimate_ne(freq1::Vector{Float64}, freq2::Vector{Float64})
    # Waples (1989) temporal method
    valid = (freq1 .> 0.05) .& (freq1 .< 0.95) .& 
            (freq2 .> 0.05) .& (freq2 .< 0.95)
    
    if sum(valid) < 100
        return NaN
    end
    
    f1 = freq1[valid]
    f2 = freq2[valid]
    
    # Standardized variance in allele frequency
    Fs = mean(((f2 - f1).^2) ./ (0.5 * (f1 + f2) .* (1 .- 0.5 * (f1 + f2))))
    
    # Correct for sampling
    S = 100  # Assumed sample size
    Fs_adj = Fs - 1/(2*S)
    
    # Ne estimate (t = 1 generation)
    Ne = 1 / (2 * Fs_adj)
    
    return max(Ne, 1.0)
end
